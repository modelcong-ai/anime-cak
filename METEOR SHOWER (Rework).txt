var NUM_STARS = 130;
var NUM_METEORS =-10000000000000000;
var NUM_CLOUDS = 6;

var stars = [];
var meteors = [];                      //CLICK TO SPAWN METEOR !!!
var clouds = [];
var particles = [];
var glows = [];

var lastBurstTime = 0;
var burstInterval =5000;
var shakeFrames = 0;

// ---- init stars
for (var i = 0; i < NUM_STARS; i++) {
    stars.push({
        x: random(width),
        y: random(height),
        r: random(1, 3),
        tw: random(80, 255)
    });
}

// ---- spawn meteor
var spawnMeteor = function() {
    var speedY = random(3, 7);
    var spawnX = random(width);
    var dir = random(1) < 0.5 ? -1 : 1;
    return {
        x: spawnX,
        y: random(-250, 0),
        vx: dir * random(0.5, 2.0),
        vy: speedY,
        len: random(40, 100),
        col: color(random(200, 255), random(180, 240), random(200, 255)),
        layer: random(1) > 0.6 ? "front" : "back",
        isCluster: false
    };
};

// ---- spawn cụm meteor
var spawnMeteorCluster = function() {
    var clusterX = random(width * 0.3, width * 0.7);
    var clusterY = random(-150, 51);
    var count = floor(random(5, 10));
    for (var i = 0; i < count; i++) {
        var dir = random(1) < 0.5 ? -1 : 1;
        meteors.push({
            x: clusterX + random(-30, 30),
            y: clusterY + random(-30, 30),
            vx: dir * random(0.5, 2.0),
            vy: random(3, 7),
            len: random(40, 100),
            col: color(random(200, 255), random(180, 240), random(200, 255)),
            layer: "front",
            isCluster: true
        });
    }
};

// ---- spawn meteor click
var spawnMeteorAt = function(px, py) {
    var speedY = random(3, 7);
    var dir = random(1) < 0.5 ? -1 : 1;
    return {
        x: px,
        y: py,
        vx: dir * random(0.5, 2.0),
        vy: speedY,
        len: random(40, 100),
        col: color(random(200, 255), random(180, 240), random(200, 255)),
        layer: "front",
        isCluster: false
    };
};

// ---- spawn particles (fire word)
var spawnParticles = function(x, y, count, col) {
    for (var i = 0; i < count; i++) {
        particles.push({
            x: x,
            y: y,
            vx: random(-3, 3),
            vy: random(-6, -2),
            life: random(40, 80),
            col: col,
            size: random(3, 7)
        });
    }
};

// ---- spawn (glow rainbow)
var spawnGlow = function(x, y, isCluster) {
    var maxR = isCluster ? random(120, 200) : random(60, 120);
    glows.push({
        x: x,
        y: y,
        r: 0,
        maxR: maxR,
        alpha: 220,
        isCluster: isCluster
    });
};

// ---- init meteors
for (var i = 0; i < NUM_METEORS; i++) {
    meteors.push(spawnMeteor());
}

// ---- init clouds
for (var i = 0; i < NUM_CLOUDS; i++) {
    clouds.push({
        x: random(width),
        y: random(height * 0.18, height * 0.48),
        speed: random(0.15, 0.6),
        size: random(70, 150)
    });
}

// ---- draw sky (gradient)
var drawSky = function() {
    for (var y = 0; y < height; y++) {
        var t = y / height;
        var c;
        if (t < 0.25) {
            c = lerpColor(color(255, 220, 100), color(255, 120, 50), map(t, 0, 0.25, 0, 1));
        } else if (t < 0.5) {
            c = lerpColor(color(255, 120, 50), color(160, 60, 160), map(t, 0.25, 0.5, 0, 1));
        } else {
            c = lerpColor(color(160, 60, 160), color(10, 20, 60), map(t, 0.5, 1, 0, 1));
        }
        stroke(c);
        line(0, y, width, y);
    }
};

// ---- stars
var drawStars = function() {
    noStroke();
    for (var i = 0; i < stars.length; i++) {
        var s = stars[i];
        fill(255, s.tw);
        ellipse(s.x, s.y, s.r, s.r);
        s.tw += random(-15, 15);
        s.tw = constrain(s.tw, 60, 255);
    }
};

// ---- moon
var drawMoon = function() {
    noStroke();
    for (var r = 200; r > 80; r -= 20) {
        fill(255, 255, 210, map(r, 80, 200, 40, 6));
        ellipse(width - 120, 100, r, r);
    }
    fill(255, 255, 220, 240);
    ellipse(width - 120, 100, 80, 80);
    fill(240, 240, 200, 230);
    ellipse(width - 120, 100, 60, 60);
};

// ---- meteor (realistic)
var drawMeteor = function(m) {
    var segs = 14, spacing = 3.5;
    for (var k = 0; k < segs; k++) {
        var x1 = m.x - m.vx * k * spacing;
        var y1 = m.y - m.vy * k * spacing;
        var x2 = m.x - m.vx * (k + 1) * spacing;
        var y2 = m.y - m.vy * (k + 1) * spacing;
        var a = map(k, 0, segs, 200, 0);
        stroke(red(m.col), green(m.col), blue(m.col), a);
        strokeWeight(map(k, 0, segs, 4.0, 0.5));
        line(x1, y1, x2, y2);
    }
    noStroke();
    fill(255, 255, 220, 230); ellipse(m.x, m.y, 10, 10);
    fill(255, 200, 150, 160); ellipse(m.x, m.y, 18, 18);
    fill(200, 200, 255, 80); ellipse(m.x, m.y, 28, 28);
};

// ---- cloud
var drawCloud = function(c) {
    noStroke();
    fill(255, 255, 255, 80);
    ellipse(c.x, c.y, c.size, c.size * 0.6);
    ellipse(c.x + 30, c.y + 10, c.size * 0.8, c.size * 0.5);
    ellipse(c.x - 30, c.y + 5, c.size * 0.7, c.size * 0.4);
};

// ---- reset meteor
var resetMeteor = function(m) {
    var newM = spawnMeteor();
    m.x = newM.x;
    m.y = newM.y;
    m.vx = newM.vx;
    m.vy = newM.vy;
    m.len = newM.len;
    m.col = newM.col;
    m.layer = newM.layer;
    m.isCluster = false;
};

// ---- update meteors
var updateAndDrawMeteors = function(layerName) {
    for (var i = 0; i < meteors.length; i++) {
        var m = meteors[i];
        if (m.layer === layerName) {
            drawMeteor(m);
            m.x += m.vx;
            m.y += m.vy;

            if (m.y > height) {
                if (m.isCluster) {
                    spawnParticles(m.x, height - 5, random(30, 45), m.col);
                    spawnGlow(m.x, height - 5, true);
                    shakeFrames = 66; // 🌟 shake  )2
                } else {
                    spawnParticles(m.x, height - 5, random(12, 20), m.col);
                    spawnGlow(m.x, height - 5, false);
                    shakeFrames =  2; // 🌟 shake (2
                }
                resetMeteor(m);
            }
        }
    }
};

// ---- update & draw particles
var updateParticles = function() {
    noStroke();
    for (var i = particles.length - 1; i >= 0; i--) {
        var p = particles[i];
        fill(red(p.col), green(p.col), blue(p.col), p.life * 3);
        ellipse(p.x, p.y, p.size, p.size);
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.12;
        p.life--;
        if (p.life <= 0) {
            particles.splice(i, 1);
        }
    }
};

// ---- update (glow rainbow)
var updateGlows = function() {
    for (var i = glows.length - 1; i >= 0; i--) {
        var g = glows[i];
        noStroke();
        for (var r = g.r; r < g.maxR; r += 15) {
            var alpha = map(r, g.r, g.maxR, g.alpha, 0);
            var col;
            if (r < g.maxR * 0.3) {
                col = color(255, 255, 255, alpha);
            } else if (r < g.maxR * 0.6) {
                col = color(255, 220, 120, alpha);
            } else {
                col = color(255, 140, 60, alpha);
            }
            fill(col);
            ellipse(g.x, g.y, r * 2, r * 2);
        }
        g.r += 5;
        g.alpha -= 8;
        if (g.alpha <= 0) {
            glows.splice(i, 1);
        }
    }
};

// ---- click: spawn meteor
mouseClicked = function() {
    meteors.push(spawnMeteorAt(mouseX, mouseY));
};

// ---- main draw
draw = function() {
    var offsetX = 0, offsetY = 0;
    if (shakeFrames > 0) {
        offsetX = random(-5, 5);
        offsetY = random(-5, 5);
        shakeFrames--;
    }

    pushMatrix();
    translate(offsetX, offsetY);

    drawSky();
    drawStars();
    updateAndDrawMeteors("back");
    drawMoon();

    for (var i = 0; i < clouds.length; i++) {
        var c = clouds[i];
        drawCloud(c);
        c.x += c.speed;
        if (c.x > width + 200) {
            c.x = -200;
            c.y = random(height * 0.18, height * 0.48);
        }
    }

    updateAndDrawMeteors("front");
    updateParticles();
    updateGlows();

    popMatrix();

    if (millis() - lastBurstTime > burstInterval) {
        if (random(1) < 0.3) {
            spawnMeteorCluster();
        }
        lastBurstTime = millis();
    }
};
