// --- Snowflakes ---
var snowflakes = [];
for (var i=0; i<250; i++) {
    snowflakes.push({
        x: random(width),
        y: random(-height, 0),
        r: random(2,6),
        speed: random(1,3),
        angle: random(0, TWO_PI)
    });
}

// --- Stars ---
var stars = [];
for (var i=0; i<100; i++) {
    stars.push({
        x: random(width),
        y: random(height/2),
        r: random(1,3),
        twinkle: random(50,255)
    });
}

// --- Sky ---
var drawSky = function() {
    for (var y=0; y<height; y++) {
        var t = y/height;
        var c = lerpColor(color(20,30,80), color(0,0,0), t);
        stroke(c);
        line(0,y,width,y);
    }
};

// --- Stars twinkle ---
var drawStars = function() {
    noStroke();
    for (var i=0; i<stars.length; i++) {
        var s = stars[i];
        fill(255, s.twinkle);
        ellipse(s.x, s.y, s.r, s.r);
        s.twinkle += random(-20,20);
        s.twinkle = constrain(s.twinkle, 50, 255);
    }
};

// --- Moon ---
var drawMoon = function() {
    var mx = width-120;
    var my = 100;
    var r = 80;
    noStroke();
    fill(255,255,200,60); ellipse(mx,my,r*2.2,r*2.2);
    fill(255,255,230,120); ellipse(mx,my,r*1.6,r*1.6);
    fill(255,255,250); ellipse(mx,my,r,r);
    fill(230,230,230,180); ellipse(mx+15,my-10,20,20);
    ellipse(mx-20,my+5,12,12);
    ellipse(mx+5,my+20,8,8);
};

// --- Ground ---
var drawGround = function() {
    noStroke();
    fill(245);
    rect(0, height-50, width, 50);
};

// --- Snowman ---
var drawSnowman = function(x,y) {
    noStroke();
    fill(255);
    ellipse(x, y, 100, 100);
    ellipse(x, y-70, 70, 70);
    ellipse(x, y-120, 50, 50);
    // eyes
    fill(0); ellipse(x-10,y-125,6,6); ellipse(x+10,y-125,6,6);
    // smile
    for (var a=-15; a<=15; a+=7) { ellipse(x+a,y-110,3,3); }
    // nose
    fill(255,120,0); triangle(x,y-120,x+25,y-115,x,y-115);
    // buttons
    fill(0); ellipse(x,y-70,6,6); ellipse(x,y-50,6,6); ellipse(x,y-30,6,6);
    // arms
    stroke(80,40,20); strokeWeight(4);
    line(x-35,y-70,x-80,y-100);
    line(x+35,y-70,x+80,y-100);
    noStroke();
    // hat
    fill(0); rect(x-25,y-145,50,20); rect(x-35,y-125,70,10);
    // scarf
    fill(200,0,0);
    rect(x-25,y-100,50,10);
    rect(x-10,y-90,10,25);
};

// --- Christmas Tree ---
var drawTree = function(x,y) {
    noStroke();
    // thân
    fill(100,50,20);
    rect(x-15,y,30,40);
    // tầng lá
    fill(0,120,40);
    triangle(x-60,y,x,y-100,x+60,y);
    triangle(x-50,y-40,x,y-130,x+50,y-40);
    triangle(x-40,y-80,x,y-160,x+40,y-80);
    // đèn nhấp nháy
    for (var i=0; i<6; i++) {
        var dx = x + random(-40,40);
        var dy = y - random(20,120);
        fill(random([color(255,0,0),color(255,255,0),color(0,200,255)]));
        ellipse(dx,dy,6,6);
    }
    // ngôi sao trên đỉnh
    fill(255,255,0);
    ellipse(x,y-170,20,20);
};

// --- Snow animation ---
var drawSnow = function() {
    noStroke();
    for (var i=0; i<snowflakes.length; i++) {
        var s = snowflakes[i];
        s.angle += 0.02;
        s.x += sin(s.angle)*0.3;
        s.y += s.speed;
        fill(255);
        ellipse(s.x,s.y,s.r,s.r);
        if (s.y > height) {
            s.y = -10;
            s.x = random(width);
            s.speed = random(1,3);
            s.r = random(2,6);
            s.angle = random(0,TWO_PI);
        }
    }
};

// --- Main draw ---
draw = function() {
    drawSky();
    drawStars();
    drawMoon();
    drawGround();
    drawTree(100,height-50);
    drawSnowman(280,height-50);
    drawSnow();
};
