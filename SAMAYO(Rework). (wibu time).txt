// ---- Particles for shooting star trail
var particles = [];

// create a particle
var makeParticle = function(x, y) {
    return {
        x: x,
        y: y,
        vx: random(-0.5, 0.5),
        vy: random(-0.5, 0.5),
        life: random(40, 80),
        size: random(1, 3)
    };
};

// update + draw particles
var updateParticles = function() {
    noStroke();
    for (var i = particles.length - 1; i >= 0; i--) {
        var p = particles[i];
        fill(255, 255, 255, p.life * 3); // fade out
        ellipse(p.x, p.y, p.size, p.size);
        p.x += p.vx;
        p.y += p.vy;
        p.life--;
        if (p.life <= 0) {
            particles.splice(i, 1);
        }
    }
};

// ---- draw multi-layer glow
var drawGlow = function(x, y) {
    noStroke();
    // Outer pink-purple glow
    for (var r = 60; r > 40; r -= 5) {
        fill(200, 100, 200, map(r, 40, 60, 30, 5));
        ellipse(x, y, r, r);
    }
    // Middle yellow glow
    for (var r = 40; r > 20; r -= 4) {
        fill(255, 240, 180, map(r, 20, 40, 40, 8));
        ellipse(x, y, r, r);
    }
    // Inner white glow
    for (var r = 20; r > 5; r -= 3) {
        fill(255, 255, 255, map(r, 5, 20, 80, 20));
        ellipse(x, y, r, r);
    }
};
