//very interesting texture ig
function interestingTexture(x, y, w, h, w2, h2) {
    
    var piixels = [];
    for(var i = 0; i < w2; i ++) {
        piixels.push([]);
        for(var j = 0; j < h2; j ++) {
            piixels[i].push(get(i, j));
        }
    }
    
    for(var i = 0; i < w2; i ++) {
        for(var j = 0; j < h2; j ++) {
            if(piixels[i][j]) {
            fill(piixels[i][j]);
            ellipse(i + random(- x, x), j + random(- y, y), w, h);
            }
            
        }
    }
    
}
